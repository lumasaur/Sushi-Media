    import Image from 'next/image'
    import { motion } from 'framer-motion'

    interface FullBleedHeroProps {
      imageSrc: string

      title: string

    interface BentoItem {
      type: 'image' | 'text' | 'stat' | 'quote'
      // For image type
      imageSrc?: string
      imageAlt?: string

      title?: string
      description?: string

      label?: string
      // For quote type

      // Common

      accentColor?: string
    }

    interface BentoGridProps {

      columns?: 2 | 3 | 4  // Default 3
      gap?: 'tight' | 'normal' | 'loose'  // 8px, 16px, 24px

      - text: card with background, icon optional
      - stat: large number, small label, accent color
