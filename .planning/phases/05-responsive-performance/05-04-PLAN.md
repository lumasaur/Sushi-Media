    ```tsx
    'use client'

    import React from 'react'
    import { useRouter, useSearchParams } from 'next/navigation'
    import { motion } from 'framer-motion'
    import {
      SlideContainer,
      ProgressBar,
      SlideCounter
    } from '@/components/presentation'
    import { useSlideNavigation, useTouchGestures } from '@/hooks'
    import {
      socialMediaSlides,
      miniStoryTitle,
      miniStorySubtitle
    } from '@/lib/slides/social-media'
    import { ChevronLeft } from 'lucide-react'

    // Animation variants (same as main presentation)
    const containerVariants = {
      hidden: { opacity: 0 },
      visible: {
        opacity: 1,
        transition: {
          staggerChildren: 0.1,
          delayChildren: 0.2
        }
      }
    }

    const itemVariants = {
      hidden: { opacity: 0, x: -20 },
      visible: {
        opacity: 1,
        x: 0,
        transition: { duration: 0.3, ease: [0.4, 0, 0.2, 1] }
      }
    }

    export default function SocialMediaMiniStory() {
      const router = useRouter()
      const searchParams = useSearchParams()
      const returnSlide = searchParams.get('returnSlide') || '4'

      const totalSlides = socialMediaSlides.length

      const {
        currentSlide,
        nextSlide,
        previousSlide,
        goToSlide,
        isFirst,
        isLast
      } = useSlideNavigation({ totalSlides })

      // Touch gestures
      useTouchGestures({
        onNavigateForward: nextSlide,
        onNavigateBackward: previousSlide
      })

      const handleBackToMain = () => {
        router.push(`/?slide=${returnSlide}`)
      }

      return (
        <div className="relative h-screen overflow-hidden">
          {/* Back to main story button */}
          <button
            onClick={handleBackToMain}
            className="fixed top-6 left-6 z-50 flex items-center gap-2 px-4 py-2 bg-sumi/80 border border-washi/20 rounded-lg text-washi hover:bg-sumi hover:border-kincha/50 transition-colors"
          >
            <ChevronLeft className="w-4 h-4" />
            <span className="text-sm">Back to Main Story</span>
          </button>

          {/* Mini-story title badge */}
          <div className="fixed top-6 right-6 z-50 px-4 py-2 bg-beni/20 border border-beni/40 rounded-lg">
            <span className="text-sm text-beni font-medium">{miniStoryTitle}</span>
          </div>

          {/* Progress indicators */}
          <ProgressBar currentSlide={currentSlide} totalSlides={totalSlides} />
          <SlideCounter currentSlide={currentSlide} totalSlides={totalSlides} />

          {/* Slides */}
          {socialMediaSlides.map((slide) => (
            <SlideContainer
              key={slide.id}
              slideNumber={slide.id}
              isActive={currentSlide === slide.id}
              background={slide.background || 'bg-sumi'}
            >
              <div className="h-full flex items-center justify-center px-12 md:px-24">
                <div className="max-w-4xl w-full">
                  <div className="text-washi">
                    <h2 className="text-4xl md:text-5xl font-bold mb-4 font-cormorant">{slide.title}</h2>
                    {slide.subtitle && (
                      <p className="text-xl md:text-2xl text-hai mb-8">{slide.subtitle}</p>
                    )}

                    <motion.ul
                      variants={containerVariants}
                      initial="hidden"
                      animate="visible"
                      className="space-y-6 text-lg"
                    >
                      {slide.bullets.map((bullet, i) => (
                        <motion.li
                          key={i}
                          variants={itemVariants}
                        >
                          {typeof bullet === 'string' ? (
                            <div className="flex items-start">
                              <span className="text-beni mr-3 mt-1">•</span>
                              <span>{bullet}</span>
                            </div>
                          ) : (
                            <div>
                              <div className="font-semibold text-kincha mb-2">{bullet.main}</div>
                              <ul className="ml-4 space-y-1 text-base text-hai">
                                {bullet.sub.map((sub, j) => (
                                  <li key={j} className="flex items-start">
                                    <span className="text-beni mr-2">•</span>
                                    <span>{sub}</span>
                                  </li>
                                ))}
                              </ul>
                            </div>
                          )}
                        </motion.li>
                      ))}
                    </motion.ul>
                  </div>
                </div>
              </div>
            </SlideContainer>
          ))}
        </div>
      )
    }
    ```
